\makeglossaries

\newacronym{UKF}{UKF}{
    Unscented Kalman Filter (сигма-точковий фільтр Калмана)
}
\newacronym{EKF}{EKF}{
    Extended Kalman Filter (розширений фільтр Калмана)
}
\newacronym{MoG2}{MoG2}{
    Mixture of Gaussians 2 (суміш гауссових розподілів)
}
\newacronym{CNN}{CNN}{
    Convolutional Neural Network (згорткова нейронна мережа)
}
\newacronym{RPN}{RPN}{
    Region Proposal Network (мережа пропозиції регіонів)
}
\newacronym{CV}{CV}{
    Computer Vision (комп'ютерний зір)
}
\newacronym{YOLO}{YOLO}{
    You Only Look Once (сімейство одноетапних моделей детекції об'єктів)
}
\newacronym{IoU}{IoU}{
    Intersection over Union (відношення перетину до об'єднання рамок)
}
\newacronym{SORT}{SORT}{
    Simple Online and Realtime Tracking (просте онлайн-відстеження в режимі реального часу)
}
\newacronym{PnP}{PnP}{
    Perspective-n-Point (задача визначення пози камери за n точками)
}
\newacronym{IMM}{IMM}{
    Interacting Multiple Model (взаємодіючі множинні моделі)
}
\newacronym{Faster R-CNN}{R-CNN}{
    Region-based Convolutional Neural Network (згорткова нейронна мережа на основі регіонів)
}
\newacronym{MOTA}{MOTA}{
    Multi-Object Tracking Accuracy (сумарна точність багатооб'єктного відстеження)
}
\newacronym{MOTP}{MOTP}{
    Multi-Object Tracking Precision (точність локалізації багатооб'єктного відстеження)
}
\newacronym{RMSE}{RMSE}{
    Root Mean Square Error (середньоквадратична похибка)
}
\newacronym{mAP}{mAP}{
    mean Average Precision (усереднена середня точність детекції)
}
\newacronym{FPS}{FPS}{
    Frames Per Second (кадрів за секунду)
}
\newacronym{API}{API}{
    Application Programming Interface (програмний інтерфейс застосунку)
}
\newacronym{GPU}{GPU}{
    Graphics Processing Unit (графічний процесор)
}
\newacronym{HUD}{HUD}{
    Heads-Up Display (накладена інформаційна панель)
}
