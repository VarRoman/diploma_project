\begin{frame}{Мета, об'єкт, задачі}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Мета:} автоматичне відновлення тривимірної траєкторії м'яча та ігрової
      статистики з відео однієї камери.\\[4pt]
      \textbf{Об'єкт:} процеси детекції та просторового відстеження об'єктів.\\[4pt]
      \textbf{Предмет:} методи 3D-трекінгу м'яча за монокулярним відео.
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Задачі:}
      \begin{enumerate}
        \item Детекція м'яча (YOLO26).
        \item Калібрування та відновлення 3D (PnP + ray casting).
        \item Згладжування траєкторії (IMM/UKF).
        \item Механізми захисту оцінки глибини.
        \item Ігрова статистика + експерименти.
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}
\note{Чітко проговорити мету/об'єкт/предмет. 5 задач — це фактично структура роботи.}

\begin{frame}{Відновлення 3D і проблема глибини}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \footnotesize
      Глибина за шириною рамки:
      \[ Z_c = \frac{f\cdot D}{w} \]
      Чутливість до похибки $w$:
      \[ \frac{\partial Z_c}{\partial w} = -\frac{f\cdot D}{w^2} \]
      \begin{itemize}
        \item Ближня зона: $\approx 0{,}14$~м/пкс.
        \item Дальня ($\sim$60~м): $\approx 2{,}3$~м/пкс.
        \item \alert{Квадратичне виродження} — корінь усіх дефектів.
      \end{itemize}
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \begin{tikzpicture}
      \begin{axis}[
        domain=4:60, samples=120,
        width=\linewidth, height=5.2cm,
        xlabel={Відстань $Z$, м}, ylabel={$|\partial Z_c/\partial w|$, м/пкс},
        xmin=0, xmax=62, ymin=0, ymax=2.6,
        grid=both, grid style={gray!20}, tick label style={font=\scriptsize},
        label style={font=\scriptsize}]
        \fill[green!12] (axis cs:0,0) rectangle (axis cs:21,2.6);
        \addplot[accent,very thick,domain=4:60] {x^2/1565};
        \draw[warn,dashed,thick] (axis cs:21,0)--(axis cs:21,2.6);
        \node[anchor=north west,font=\tiny,good,align=left] at (axis cs:1,2.5)
          {робоча зона};
        \node[warn,anchor=south west,font=\tiny] at (axis cs:21.4,1.5) {стеля 21\,м};
      \end{axis}
      \end{tikzpicture}\\
      {\scriptsize Похибка глибини зростає квадратично з відстанню}
    \end{column}
  \end{columns}
\end{frame}
\note{Ключовий слайд. Пояснити формулу, чому глибина виродна на дальньому боці. Це мотивує всі механізми захисту.}

\begin{frame}{Трекер: IMM/UKF + захист глибини}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\footnotesize
        \item \textbf{UKF}: нелінійне згладжування 9D-стану (позиція, швидкість, прискорення).
        \item \textbf{IMM}: ансамбль моделей руху (балістична / ударна / відскокова).
        \item \textbf{Гейт A}: інфляція $\sigma_Z^2$ при великій глибинній нев'язці.
        \item \textbf{Коастинг}: інерційний прогноз із clamp $Z\in[-3,21]$.
        \item \textbf{Far-Z стеля} детекції 21~м — відсікає фоновий м'яч.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}

\begin{frame}{Висновки}
  \begin{itemize}
    \item Побудовано \alert{наскрізний автоматичний} конвеєр відео~$\rightarrow$~3D-статистика у реальному часі.
    \item Детектор YOLO26: mAP@0.5~$=0{,}857$ на власному датасеті.
    \item Розроблено й кількісно підтверджено \alert{комплекс механізмів захисту глибини}
          (стрибки $60\to6$, фрагментація $123\to20$).
    \item $g_{\text{відн}}\approx 9{,}1$~м/с$^2$ — геометрична адекватність траєкторій.
    \item \textbf{Практична цінність:} автоматична просторово-часова статистика гри з однієї камери.
  \end{itemize}
